cmake_minimum_required(VERSION 3.22)
project(ct3_xbox_recomp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(CT2_VERBOSE_LOG "Enable verbose diagnostic logging" OFF)
option(CT2_ENABLE_PRELOAD "Enable background shader/pipeline preloading" ON)
option(CT2_BUILD_DESKTOP_STUB "Build host executable target on Linux desktop" OFF)
option(CT3_ENABLE_XBE_SDK "Compile and link portable XboxXBERecomp sdk/runtime + 128 lifted partitions" OFF)

# Strip unused third-party library subsystems when vendored under third_party/ or sdk/
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)

set(CT3_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk")
set(CT3_GEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated/game")
set(CT3_USE_SDK OFF)
if(CT3_ENABLE_XBE_SDK AND EXISTS "${CT3_SDK_DIR}/runtime" AND EXISTS "${CT3_GEN_DIR}/x86_func_mapping.cpp")
    set(CT3_USE_SDK ON)
    message(STATUS "CT3: Full XboxXBERecomp SDK + 128 lifted Crazy Taxi 3 partitions ENABLED")
else()
    message(STATUS "CT3: Building standalone Vulkan diagnostic runtime (sdk/ or generated/game not enabled)")
endif()

# ---------------------------------------------------------------------------
# Recompiled Xbox x86 partitions (128 TUs generated by XboxXBERecomp)
# Always compiled with -O2 -w -fno-strict-aliasing -fsigned-char even in debug
# ---------------------------------------------------------------------------
if(CT3_USE_SDK)
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/config/game.toml" CT3_PARTITIONS_LINE REGEX "^partitions *=")
    string(REGEX MATCH "[0-9]+" CT3_PARTITIONS "${CT3_PARTITIONS_LINE}")
    if(NOT CT3_PARTITIONS)
        set(CT3_PARTITIONS 128)
    endif()
    math(EXPR CT3_LAST_PART "${CT3_PARTITIONS} - 1")
    set(CT2_RECOMP_SOURCES "${CT3_GEN_DIR}/x86_func_mapping.cpp")
    foreach(i RANGE ${CT3_LAST_PART})
        list(APPEND CT2_RECOMP_SOURCES "${CT3_GEN_DIR}/x86_recomp.${i}.cpp")
    endforeach()
else()
    file(GLOB CT2_RECOMP_SOURCES CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/*.cpp"
    )
endif()

add_library(ct2_recomp_code STATIC ${CT2_RECOMP_SOURCES})
target_include_directories(ct2_recomp_code PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
)
if(CT3_USE_SDK)
    target_include_directories(ct2_recomp_code PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/native/compat"
        "${CT3_SDK_DIR}/harness/compat"
        "${CT3_SDK_DIR}/runtime"
        "${CT3_GEN_DIR}"
    )
endif()
target_compile_options(ct2_recomp_code PRIVATE
    -O2
    -w
    -fno-strict-aliasing
    -fsigned-char
)

# ---------------------------------------------------------------------------
# Native Xbox host runtime + mobile Vulkan renderer
# ---------------------------------------------------------------------------
set(CT2_HOST_SOURCES
    src/host/memory.cpp
    src/host/xiso.cpp
    src/host/xbe_loader.cpp
    src/host/rom_vfs.cpp
    src/host/save_guard.cpp
    src/host/hang_watchdog.cpp
    src/kernel/xbox_kernel.cpp
    src/gfx/render_thread.cpp
    src/d3d/d3d8_bridge.cpp
    src/game/game_loop.cpp
    native/stats.cpp
    native/dynamic_resolution.cpp
    native/vk_shader.cpp
    native/vulkan.cpp
    native/ct3_app_host.cpp
    native/ct3_gpu_backend.cpp
)

if(CT3_USE_SDK)
    file(GLOB CT3_SDK_RUNTIME_SOURCES CONFIGURE_DEPENDS
        "${CT3_SDK_DIR}/runtime/*/*.cpp"
    )
    # Exclude Windows/x86-SSE2-only SDK host layers (replaced by Android native/ct3_app_host.cpp + ct3_gpu_backend.cpp)
    list(FILTER CT3_SDK_RUNTIME_SOURCES EXCLUDE REGEX
        "/runtime/app/|/core/input\\.cpp$|/gpu/(d3d12|render_thread)\\.cpp$|/audio/xaudio2\\.cpp$"
    )
    list(APPEND CT2_HOST_SOURCES ${CT3_SDK_RUNTIME_SOURCES})
endif()

if(ANDROID)
    list(APPEND CT2_HOST_SOURCES native/android_jni.cpp)
    add_library(main SHARED ${CT2_HOST_SOURCES})
else()
    if(CT2_BUILD_DESKTOP_STUB)
        add_executable(main ${CT2_HOST_SOURCES} src/host/desktop_main.cpp)
    else()
        add_library(main STATIC ${CT2_HOST_SOURCES})
    endif()
endif()

target_include_directories(main PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${CMAKE_CURRENT_SOURCE_DIR}/native"
)
if(CT3_USE_SDK)
    target_include_directories(main PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/native/compat"
        "${CT3_SDK_DIR}/harness/compat"
        "${CT3_SDK_DIR}/runtime"
        "${CT3_GEN_DIR}"
    )
    target_compile_definitions(main PRIVATE CT3_ENABLE_XBE_SDK=1)
endif()

# Match x86 signed char and aliasing semantics on ARM64 host sources
target_compile_options(main PRIVATE
    -fno-strict-aliasing
    -fsigned-char
    -Wall
    -Wextra
    -Wno-unused-parameter
    -Wno-sign-compare
    -Wno-missing-field-initializers
)

# Force-include minimal NDK compat header into host sources ONLY (never generated recomp code)
target_compile_options(main PRIVATE
    -include "${CMAKE_CURRENT_SOURCE_DIR}/native/compat/crt.h"
)

if(CT2_VERBOSE_LOG)
    target_compile_definitions(main PRIVATE CT2_VERBOSE_LOG=1 XBE_VERBOSE_LOG=1)
endif()

if(CT2_ENABLE_PRELOAD)
    target_compile_definitions(main PRIVATE CT2_ENABLE_PRELOAD=1)
endif()

target_link_libraries(main PRIVATE ct2_recomp_code)

find_package(Threads REQUIRED)
target_link_libraries(main PRIVATE Threads::Threads)

if(ANDROID)
    find_library(ANDROID_LIB android REQUIRED)
    find_library(LOG_LIB log REQUIRED)
    find_library(VULKAN_LIB vulkan)

    target_link_libraries(main PRIVATE ${ANDROID_LIB} ${LOG_LIB} dl)
    if(VULKAN_LIB)
        target_link_libraries(main PRIVATE ${VULKAN_LIB})
    endif()

    # Android 15+ 16 KB page-size alignment (verify with: readelf -lW libmain.so)
    target_link_options(main PRIVATE -Wl,-z,max-page-size=16384)
else()
    find_package(Vulkan QUIET)
    if(Vulkan_FOUND)
        target_link_libraries(main PRIVATE Vulkan::Vulkan)
    endif()
    target_link_libraries(main PRIVATE dl)
endif()
